def lookup_term_impl(term: str) -> str:
    """Look up a term using Free Dictionary API (no API key). Returns first definition or an error string."""
    try:
        url = f"https://api.dictionaryapi.dev/api/v2/entries/en/{quote(term.strip())}"
        with urlopen(Request(url, headers={"User-Agent": "Layer2Demo/1.0"})) as r:
            data = json.loads(r.read().decode())
        if not data or not isinstance(data, list):
            return f'No definition found for "{term}"'
        # First entry, first meaning, first definition
        entry = data[0]
        meanings = entry.get("meanings") or []
        if not meanings:
            return f'No definition found for "{term}"'
        definitions = meanings[0].get("definitions") or []
        if not definitions:
            return f'No definition found for "{term}"'
        return definitions[0].get("definition", "(no definition text)")
    except Exception as e:
        return f'Lookup error for "{term}": {e}'
